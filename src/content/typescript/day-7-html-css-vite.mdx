# What is Vite?

Vite is a build tool that provides a fast development environment for modern web projects. Unlike traditional bundlers, Vite serves your code via native ES modules during development, making it incredibly fast.

Key features:
- **Instant server start** - No bundling during development
- **Hot Module Replacement (HMR)** - Changes update instantly in the browser
- **Built-in TypeScript support** - No additional configuration needed
- **Optimized builds** - Uses Rollup for production builds

# Setting Up a Vite Project

Let's create a new Vite project with TypeScript:

```bash
# Create a new Vite project
bun create vite my-web-app --template vanilla-ts
cd my-web-app

# Install dependencies
bun install

# Start development server
bun run dev
```

This creates a project structure like:

```
my-web-app/
├── src/
│   ├── main.ts
│   ├── style.css
│   └── vite-env.d.ts
├── index.html
├── package.json
├── tsconfig.json
└── vite.config.ts
```

# Connecting TypeScript

Now let's add interactivity with TypeScript:

```ts
// src/main.ts
interface TodoItem {
  id: number;
  text: string;
  completed: boolean;
}

class TodoApp {
  private todos: TodoItem[] = [];
  private nextId = 1;
  private todoList: HTMLElement;
  private todoInput: HTMLInputElement;

  constructor() {
    this.todoList = document.getElementById("todo-list")!;
    this.todoInput = document.getElementById("todo-input") as HTMLInputElement;
    
    this.setupEventListeners();
    this.render();
  }

  private setupEventListeners(): void {
    const addButton = document.getElementById("add-todo");
    addButton?.addEventListener("click", () => this.addTodo());
    
    this.todoInput.addEventListener("keypress", (e) => {
      if (e.key === "Enter") {
        this.addTodo();
      }
    });
  }

  private addTodo(): void {
    const text = this.todoInput.value.trim();
    if (!text) return;

    const todo: TodoItem = {
      id: this.nextId++,
      text,
      completed: false
    };

    this.todos.push(todo);
    this.todoInput.value = "";
    this.render();
  }

  private toggleTodo(id: number): void {
    const todo = this.todos.find(t => t.id === id);
    if (todo) {
      todo.completed = !todo.completed;
      this.render();
    }
  }

  private deleteTodo(id: number): void {
    this.todos = this.todos.filter(t => t.id !== id);
    this.render();
  }

  private render(): void {
    this.todoList.innerHTML = "";

    this.todos.forEach(todo => {
      const li = document.createElement("li");
      li.className = `todo-item ${todo.completed ? "completed" : ""}`;
      
      li.innerHTML = `
        <span class="todo-text">${todo.text}</span>
        <div class="todo-actions">
          <button class="toggle-btn" data-id="${todo.id}">
            ${todo.completed ? "Undo" : "Done"}
          </button>
          <button class="delete-btn" data-id="${todo.id}">Delete</button>
        </div>
      `;

      // Add event listeners
      const toggleBtn = li.querySelector(".toggle-btn") as HTMLButtonElement;
      const deleteBtn = li.querySelector(".delete-btn") as HTMLButtonElement;
      
      toggleBtn.addEventListener("click", () => this.toggleTodo(todo.id));
      deleteBtn.addEventListener("click", () => this.deleteTodo(todo.id));

      this.todoList.appendChild(li);
    });
  }
}

// Initialize the app when DOM is loaded
document.addEventListener("DOMContentLoaded", () => {
  new TodoApp();
});
```

```ts

  });
```
